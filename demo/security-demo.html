    🔐 Input Validation Implementation

    function validateMessage(message) { if (typeof message !== 'string') { throw new Error('Invalid message type'); } if (message.length === 0) { throw new Error('Empty message'); } if (message.length > 10000) { throw new Error('Message too long'); } // Remove dangerous content const sanitized = message .replace(/<script\b[^<]*<\/script>/gi, '') .replace(/javascript:/gi, '') .replace(/on\w+\s*=/gi, '') .trim(); return sanitized; }
